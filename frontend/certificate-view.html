  </main>


  <script>

    "use strict";


    const result = document.getElementById("result");

    const number = new URLSearchParams(window.location.search).get("number");


    function showError(message) {

      const box = document.createElement("div");

      box.className = "error";

      box.textContent = message;

      result.replaceChildren(box);

    }


    async function loadCertificate() {

      if (!number) {

        showError("Certificate number is missing from the link.");

        return;

      }


      try {

        const response = await fetch(

          "http://localhost:8000/api/v1/catalog/certificates/" +

          encodeURIComponent(number) + "/verify"

        );


        if (!response.ok) {

          throw new Error("Certificate could not be verified.");

        }


        const certificate = await response.json();

        const card = document.createElement("article");

        card.className = "certificate";


        function add(tag, className, text) {

          const node = document.createElement(tag);

          node.className = className;

          node.textContent = text;

          card.append(node);

        }


        add("div", "brand", "SKILLPULSE");

        add(

          "div",

          "eyebrow",

          certificate.is_valid ? "Verified course completion" : "Certificate revoked"

        );

        add("h1", "", "Certificate of Completion");

        add("p", "recipient", "This certificate is awarded to");

        add("h2", "name", certificate.learner_name);

        add("p", "recipient", "for completing");

        add("p", "award", certificate.certification_name);

        add("p", "course", certificate.course_title);


        const details = document.createElement("div");

        details.className = "details";


        [

          ["Certificate number", certificate.certificate_number],

          ["Date issued", new Date(certificate.issued_at).toLocaleDateString()],

          ["Status", certificate.is_valid ? "Valid" : "Revoked"]

        ].forEach(([label, value]) => {

          const item = document.createElement("div");

          item.textContent = `${label}: ${value}`;

          details.append(item);

        });


        card.append(details);


        add(

          "p",

          "disclaimer",

          "This is a SkillPulse course completion certificate. It is not an AWS certification and does not represent an AWS credential."

        );


        result.replaceChildren(card);

      } catch (error) {

        showError(error.message);

      }

    }


    loadCertificate();

  </script>
